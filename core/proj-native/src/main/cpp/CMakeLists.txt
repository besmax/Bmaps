cmake_minimum_required(VERSION 3.22)
project(bmaps_projection LANGUAGES C CXX)

if(NOT EXISTS "${BMAPS_PROJ_SOURCE}/CMakeLists.txt" OR NOT EXISTS "${BMAPS_SQLITE_SOURCE}/sqlite3.c")
    message(FATAL_ERROR "Catalog-pinned PROJ and SQLite sources are required")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

# Isolate the SQLite C ABI from Room, platform SQLite and other static dependencies.
file(READ "${BMAPS_SQLITE_SOURCE}/sqlite3.h" SQLITE_HEADER)
string(REGEX MATCHALL "sqlite3_[A-Za-z0-9_]+" SQLITE_NAMES "${SQLITE_HEADER}")
list(REMOVE_DUPLICATES SQLITE_NAMES)
set(SQLITE_RENAMES "#define sqlite3 bmaps_proj_sqlite3\n")
foreach(NAME IN LISTS SQLITE_NAMES)
    string(APPEND SQLITE_RENAMES "#define ${NAME} bmaps_proj_${NAME}\n")
endforeach()
set(SQLITE_PREFIX_HEADER "${CMAKE_CURRENT_BINARY_DIR}/sqlite_prefix.h")
file(WRITE "${SQLITE_PREFIX_HEADER}" "${SQLITE_RENAMES}")
add_library(bmaps_proj_sqlite STATIC "${BMAPS_SQLITE_SOURCE}/sqlite3.c")
target_include_directories(bmaps_proj_sqlite PUBLIC "${BMAPS_SQLITE_SOURCE}")
target_compile_options(bmaps_proj_sqlite PRIVATE -include "${SQLITE_PREFIX_HEADER}")
target_compile_definitions(bmaps_proj_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_APPS OFF CACHE BOOL "" FORCE)
set(ENABLE_CURL OFF CACHE BOOL "" FORCE)
set(ENABLE_TIFF OFF CACHE BOOL "" FORCE)
set(EMBED_RESOURCE_FILES ON CACHE BOOL "" FORCE)
set(USE_ONLY_EMBEDDED_RESOURCE_FILES ON CACHE BOOL "" FORCE)
set(EMBED_PROJ_DATA_PATH OFF CACHE BOOL "" FORCE)
set(PROJ_OBJECT_LIBRARIES_POSITION_INDEPENDENT_CODE ON CACHE BOOL "" FORCE)
set(NLOHMANN_JSON_ORIGIN internal CACHE STRING "" FORCE)
find_program(EXE_SQLITE3 sqlite3 NO_CMAKE_FIND_ROOT_PATH REQUIRED)
add_subdirectory("${BMAPS_PROJ_SOURCE}" proj EXCLUDE_FROM_ALL)
target_compile_options(proj PRIVATE -include "${SQLITE_PREFIX_HEADER}")
set_target_properties(proj PROPERTIES FRAMEWORK FALSE ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

if(ANDROID)
    add_library(bmaps_proj SHARED bmaps_proj.c bmaps_proj_jni.c)
    target_link_options(bmaps_proj PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
    target_link_libraries(bmaps_proj PRIVATE log)
else()
    add_library(bmaps_proj STATIC bmaps_proj.c)
endif()
target_include_directories(bmaps_proj PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(bmaps_proj PRIVATE PROJ::proj)
add_dependencies(bmaps_proj bmaps_proj_sqlite)
target_compile_options(bmaps_proj PRIVATE -Wall -Wextra -Werror)
